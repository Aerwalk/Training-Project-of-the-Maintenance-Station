````java
package org.example.sto;

import org.example.sto.cli.ConsoleMenu;
import org.example.sto.cli.DemoRunner;
import org.example.sto.config.AppConfig;
import org.example.sto.config.DbSettings;
import org.example.sto.db.ConnectionManager;
import org.example.sto.exception.DatabaseException;
import org.example.sto.model.OrderSummary;
import org.example.sto.repository.CarRepository;
import org.example.sto.repository.ClientRepository;
import org.example.sto.repository.MasterRepository;
import org.example.sto.repository.OrderRepository;
import org.example.sto.repository.ServiceRepository;
import org.example.sto.repository.UserRepository;
import org.example.sto.service.AuthService;
import org.example.sto.service.CatalogService;
import org.example.sto.service.OrderService;

import java.io.FileDescriptor;
import java.io.FileOutputStream;
import java.io.PrintStream;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.List;

/**
 * Точка входа учебного приложения «СТО».
 *
 * <p>Режимы:
 * <pre>
 *   java -jar sto-db-module.jar              интерактивное меню
 *   java -jar sto-db-module.jar --demo       автоматический smoke-сценарий
 *   java -jar sto-db-module.jar --check      проверка соединения и версии PostgreSQL
 *   java -jar sto-db-module.jar --list-orders список заявок без входа
 *   java -jar sto-db-module.jar --help       справка
 * </pre>
 */
public final class StoApplication {

    private StoApplication() {
    }

    public static void main(String[] args) {
        configureUtf8Console();

        boolean demo = hasArg(args, "--demo");
        boolean check = hasArg(args, "--check");
        boolean listOrders = hasArg(args, "--list-orders");
        boolean help = hasArg(args, "--help") || hasArg(args, "-h");

        if (help) {
            printHelp();
            return;
        }

        DbSettings settings;
        try {
            settings = AppConfig.load().dbSettings();
        } catch (RuntimeException e) {
            System.err.println("Ошибка конфигурации: " + e.getMessage());
            System.exit(2);
            return;
        }

        try (ConnectionManager connectionManager = new ConnectionManager(settings)) {
            UserRepository userRepository = new UserRepository(connectionManager);
            ClientRepository clientRepository = new ClientRepository(connectionManager);
            CarRepository carRepository = new CarRepository(connectionManager);
            MasterRepository masterRepository = new MasterRepository(connectionManager);
            ServiceRepository serviceRepository = new ServiceRepository(connectionManager);
            OrderRepository orderRepository = new OrderRepository(connectionManager);

            AuthService authService = new AuthService(userRepository);
            CatalogService catalogService = new CatalogService(
                    clientRepository, carRepository, serviceRepository, masterRepository, orderRepository);
            OrderService orderService = new OrderService(
                    orderRepository, clientRepository, carRepository, masterRepository, serviceRepository);

            if (check) {
                String version = connectionManager.ping();
                System.out.println("Подключение успешно. " + settings.toSafeString());
                System.out.println("PostgreSQL: " + version);
                return;
            }

            if (listOrders) {
                printOrders(orderService.listAll());
                return;
            }

            if (demo) {
                new DemoRunner(authService, catalogService, orderService, connectionManager).run();
                return;
            }

            new ConsoleMenu(authService, catalogService, orderService, connectionManager, System.out).run();
        } catch (DatabaseException e) {
            System.err.println("Ошибка БД: " + e.getMessage());
            if (e.getSqlState() != null) {
                System.err.println("SQLState: " + e.getSqlState());
            }
            System.exit(1);
        } catch (RuntimeException e) {
            System.err.println("Непредвиденная ошибка: " + e.getMessage());
            e.printStackTrace(System.err);
            System.exit(1);
        }
    }

    private static boolean hasArg(String[] args, String expected) {
        return Arrays.asList(args).contains(expected);
    }

    /**
     * Единая кодировка вывода, чтобы кириллица корректно отображалась
     * в консоли Windows, Git Bash и IDE.
     */
    private static void configureUtf8Console() {
        System.setOut(new PrintStream(new FileOutputStream(FileDescriptor.out), true, StandardCharsets.UTF_8));
        System.setErr(new PrintStream(new FileOutputStream(FileDescriptor.err), true, StandardCharsets.UTF_8));
    }

    private static void printOrders(List<OrderSummary> orders) {
        System.out.printf("%-4s %-32s %-13s %-13s %12s%n", "ID", "Клиент", "Авто", "Статус", "Стоимость");
        for (OrderSummary order : orders) {
            System.out.printf("%-4d %-32s %-13s %-13s %12s%n",
                    order.orderId(), order.client(), order.car(), order.status(), order.totalCost());
        }
    }

    private static void printHelp() {
        System.out.println("""
                Модуль доступа к БД СТО (PostgreSQL).
                Параметры подключения: src/main/resources/application.properties
                или -Dconfig.file=<путь>, переменные STO_DB_URL / STO_DB_USERNAME / STO_DB_PASSWORD.

                Режимы:
                  --demo          автоматический сквозной сценарий
                  --check         проверить соединение и версию PostgreSQL
                  --list-orders   вывести заявки без аутентификации
                  --help, -h      эта справка

                Тестовые учётные записи:
                  admin/admin123, manager/manager123, master/master123, client/client123
                """);
    }
}
````

````java
package org.example.sto.cli;

import org.example.sto.db.ConnectionManager;
import org.example.sto.exception.DatabaseException;
import org.example.sto.exception.EntityNotFoundException;
import org.example.sto.exception.ValidationException;
import org.example.sto.model.Car;
import org.example.sto.model.Client;
import org.example.sto.model.ClientOrder;
import org.example.sto.model.Master;
import org.example.sto.model.Order;
import org.example.sto.model.OrderStatus;
import org.example.sto.model.OrderStatusLogEntry;
import org.example.sto.model.OrderSummary;
import org.example.sto.model.Service;
import org.example.sto.model.User;
import org.example.sto.service.AuthService;
import org.example.sto.service.CatalogService;
import org.example.sto.service.OrderService;

import java.io.BufferedReader;
import java.io.Console;
import java.io.InputStreamReader;
import java.io.PrintStream;
import java.math.BigDecimal;
import java.nio.charset.StandardCharsets;
import java.util.List;
import java.util.Set;

/**
 * Консольный интерфейс учебного приложения СТО.
 */
public final class ConsoleMenu {

    private static final Set<String> STAFF_ROLES = Set.of("admin", "manager", "master");

    private final AuthService authService;
    private final CatalogService catalogService;
    private final OrderService orderService;
    private final ConnectionManager connectionManager;

    private final BufferedReader reader;
    private final PrintStream out;

    public ConsoleMenu(AuthService authService,
                       CatalogService catalogService,
                       OrderService orderService,
                       ConnectionManager connectionManager,
                       PrintStream out) {
        this.authService = authService;
        this.catalogService = catalogService;
        this.orderService = orderService;
        this.connectionManager = connectionManager;
        this.out = out;
        this.reader = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
    }

    public void run() {
        printBanner();
        User user = login();
        if (user == null) {
            return;
        }

        out.println();
        out.printf("Вы вошли как %s (%s, роль: %s)%n", user.fullName(), user.username(), user.role());

        boolean staff = STAFF_ROLES.contains(user.role());
        boolean running = true;
        while (running) {
            printMenu(staff, user.role());
            int choice = readInt("Выберите пункт: ", -1);
            try {
                switch (choice) {
                    case 1 -> showOrders();
                    case 2 -> { if (staff) showClients(); else unknown(); }
                    case 3 -> { if (staff) showClientCars(); else unknown(); }
                    case 4 -> showServices();
                    case 5 -> { if (staff) createOrder(user); else unknown(); }
                    case 6 -> { if (staff) addService(); else unknown(); }
                    case 7 -> { if (staff) closeOrder(); else unknown(); }
                    case 8 -> showTotal();
                    case 9 -> showClientOrders();
                    case 10 -> { if (staff) showStatusLog(); else unknown(); }
                    case 11 -> ping();
                    case 0 -> running = false;
                    default -> out.println("Неизвестный пункт меню.");
                }
            } catch (DatabaseException e) {
                out.println("Ошибка БД: " + e.getMessage());
                if (e.getSqlState() != null) {
                    out.println("SQLState: " + e.getSqlState() + ", операция: " + e.getOperation());
                }
            } catch (ValidationException | EntityNotFoundException e) {
                out.println("Ошибка: " + e.getMessage());
            }
        }
        out.println("Работа завершена.");
    }

    private User login() {
        for (int attempt = 1; attempt <= 3; attempt++) {
            out.print("Логин: ");
            out.flush();
            String username;
            try {
                username = reader.readLine();
            } catch (Exception e) {
                return null;
            }
            if (username == null) {
                return null;
            }
            char[] password = readPassword();
            try {
                return authService.login(username, password);
            } catch (RuntimeException e) {
                out.println("Не удалось войти: " + e.getMessage());
                out.println("Попытка " + attempt + " из 3.");
            }
        }
        out.println("Превышено количество попыток входа.");
        return null;
    }

    private char[] readPassword() {
        // Приглашение печатаем через PrintStream с UTF-8: readPassword(prompt)
        // пишет напрямую в консоль в кодировке ОС, из-за чего кириллица ломается.
        out.print("Пароль: ");
        out.flush();
        Console console = System.console();
        if (console != null) {
            char[] password = console.readPassword();
            return password == null ? new char[0] : password;
        }
        try {
            String line = reader.readLine();
            return line == null ? new char[0] : line.toCharArray();
        } catch (Exception e) {
            return new char[0];
        }
    }

    private void printBanner() {
        out.println("==============================================================");
        out.println("  Модуль доступа к БД СТО (PostgreSQL + JDBC + HikariCP)");
        out.println("  Учебная практика УП.01.02, ПМ.01");
        out.println("==============================================================");
    }

    private void printMenu(boolean staff, String role) {
        out.println();
        out.println("--- Главное меню (" + role + ") ---");
        out.println(" 1. Список заявок (представление v_order_total)");
        if (staff) {
            out.println(" 2. Список клиентов");
            out.println(" 3. Автомобили клиента");
        }
        out.println(" 4. Список услуг");
        if (staff) {
            out.println(" 5. Создать заявку (create_order)");
            out.println(" 6. Добавить услугу (add_service_to_order)");
            out.println(" 7. Закрыть заявку (close_order)");
        }
        out.println(" 8. Рассчитать стоимость (calculate_order_total)");
        out.println(" 9. Заявки клиента (get_client_orders)");
        if (staff) {
            out.println("10. Журнал статусов");
        }
        out.println("11. Проверить соединение с БД");
        out.println(" 0. Выход");
    }

    private void showOrders() {
        List<OrderSummary> orders = orderService.listAll();
        out.println();
        out.printf("%-4s %-32s %-13s %-13s %12s%n", "ID", "Клиент", "Авто", "Статус", "Стоимость");
        out.println("-".repeat(80));
        for (OrderSummary order : orders) {
            out.printf("%-4d %-32s %-13s %-13s %12s%n",
                    order.orderId(), order.client(), order.car(), order.status(), order.totalCost());
        }
        if (orders.isEmpty()) {
            out.println("Заявок нет.");
        }
    }

    private void showClients() {
        List<Client> clients = catalogService.clients();
        out.println();
        out.printf("%-4s %-32s %-18s %-24s%n", "ID", "ФИО", "Телефон", "E-mail");
        out.println("-".repeat(82));
        for (Client client : clients) {
            out.printf("%-4d %-32s %-18s %-24s%n",
                    client.id(), client.fullName(), client.phone(),
                    client.email() == null ? "—" : client.email());
        }
    }

    private void showClientCars() {
        showClients();
        int clientId = readInt("ID клиента: ", -1);
        List<Car> cars = catalogService.clientCars(clientId);
        out.println();
        out.printf("%-4s %-12s %-12s %-6s %-12s %-18s%n", "ID", "Марка", "Модель", "Год", "Номер", "VIN");
        out.println("-".repeat(70));
        for (Car car : cars) {
            out.printf("%-4d %-12s %-12s %-6s %-12s %-18s%n",
                    car.id(), car.brand(), car.model(),
                    car.year() == null ? "—" : car.year().toString(),
                    car.plateNum(), car.vin() == null ? "—" : car.vin());
        }
        if (cars.isEmpty()) {
            out.println("У клиента нет автомобилей.");
        }
    }

    private void showServices() {
        List<Service> services = catalogService.services();
        out.println();
        out.printf("%-4s %-34s %10s %8s%n", "ID", "Наименование", "Цена", "Мин.");
        out.println("-".repeat(62));
        for (Service service : services) {
            out.printf("%-4d %-34s %10s %8s%n",
                    service.id(), service.name(), service.price(),
                    service.durationMin() == null ? "—" : service.durationMin());
        }
    }

    private void createOrder(User user) {
        showClients();
        int clientId = readInt("ID клиента: ", -1);
        List<Car> cars = catalogService.clientCars(clientId);
        if (cars.isEmpty()) {
            out.println("У клиента нет автомобилей. Сначала добавьте автомобиль.");
            return;
        }
        out.printf("%-4s %-12s %-12s %-12s%n", "ID", "Марка", "Модель", "Номер");
        for (Car car : cars) {
            out.printf("%-4d %-12s %-12s %-12s%n", car.id(), car.brand(), car.model(), car.plateNum());
        }
        int carId = readInt("ID автомобиля: ", -1);

        List<Master> masters = catalogService.masters();
        out.println("Мастера:");
        for (Master master : masters) {
            out.printf("  %d — %s (%s)%n", master.id(), master.fullName(),
                    master.active() ? "активен" : "неактивен");
        }
        int masterId = readInt("ID мастера (0 — без мастера): ", 0);

        out.print("Описание неисправности: ");
        out.flush();
        String description = readLine();

        int orderId = orderService.createOrder(clientId, carId, masterId == 0 ? null : masterId, description);
        out.println("Создана заявка №" + orderId + " (пользователь: " + user.username() + ")");
    }

    private void addService() {
        showServices();
        int orderId = readInt("ID заявки: ", -1);
        int serviceId = readInt("ID услуги: ", -1);
        int quantity = readInt("Количество (1..100): ", 1);
        orderService.addService(orderId, serviceId, quantity);
        out.println("Услуга добавлена в заявку №" + orderId);
    }

    private void closeOrder() {
        int orderId = readInt("ID заявки для закрытия: ", -1);
        orderService.closeOrder(orderId);
        BigDecimal total = orderService.calculateTotal(orderId);
        out.println("Заявка №" + orderId + " завершена. Итоговая стоимость: " + total);
    }

    private void showTotal() {
        int orderId = readInt("ID заявки: ", -1);
        out.println("Стоимость заявки №" + orderId + ": " + orderService.calculateTotal(orderId));
    }
````

````java
package org.example.sto.db;

import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;
import org.example.sto.config.DbSettings;
import org.example.sto.exception.DatabaseException;
import org.example.sto.exception.RetryExhaustedException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.sql.Connection;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;

/**
 * Центральная точка доступа к БД.
 *
 * <p>Отвечает за:
 * <ul>
 *     <li>пул соединений HikariCP;</li>
 *     <li>повторные попытки получить соединение при временной недоступности БД;</li>
 *     <li>транзакции (commit/rollback);</li>
 *     <li>трансляцию SQLException в предметные исключения.</li>
 * </ul>
 *
 * <p>Экземпляр класса потокобезопасен и должен использоваться как singleton
 * на всё приложение.
 */
public final class ConnectionManager implements AutoCloseable {

    private static final Logger log = LoggerFactory.getLogger(ConnectionManager.class);

    private final HikariDataSource dataSource;
    private final DbSettings settings;
    private final RetryPolicy retryPolicy;

    public ConnectionManager(DbSettings settings) {
        this(settings, new RetryPolicy(
                settings.retryMaxAttempts(),
                settings.retryInitialDelayMs(),
                settings.retryMultiplier(),
                settings.retryMaxDelayMs()));
    }

    public ConnectionManager(DbSettings settings, RetryPolicy retryPolicy) {
        this.settings = settings;
        this.retryPolicy = retryPolicy;

        HikariConfig config = new HikariConfig();
        config.setPoolName("sto-pool");
        config.setJdbcUrl(settings.url());
        config.setUsername(settings.username());
        config.setPassword(settings.password());
        config.setDriverClassName(settings.driver());
        config.setMaximumPoolSize(settings.maximumPoolSize());
        config.setMinimumIdle(settings.minimumIdle());
        config.setConnectionTimeout(settings.connectionTimeoutMs());
        config.setValidationTimeout(settings.validationTimeoutMs());
        config.setIdleTimeout(settings.idleTimeoutMs());
        config.setMaxLifetime(settings.maxLifetimeMs());
        config.setAutoCommit(true);
        // Не падаем на старте, если БД ещё не поднялась: приложение выполнит retry
        config.setInitializationFailTimeout(-1);
        config.setRegisterMbeans(false);

        this.dataSource = new HikariDataSource(config);
        log.info("Пул соединений создан: {}", settings.toSafeString());
    }

    public DbSettings settings() {
        return settings;
    }

    /**
     * Получает соединение с повторными попытками.
     *
     * @throws RetryExhaustedException если все попытки исчерпаны
     */
    public Connection getConnection() {
        SQLException lastError = null;
        int attempts = retryPolicy.maxAttempts();

        for (int attempt = 1; attempt <= attempts; attempt++) {
            try {
                Connection connection = dataSource.getConnection();
                if (isValid(connection)) {
                    return connection;
                }
                connection.close();
                lastError = new SQLException("Соединение не прошло проверку isValid()", "08006");
            } catch (SQLException e) {
                lastError = e;
                log.warn("Попытка подключения {}/{} не удалась: {}", attempt, attempts, e.getMessage());
            }

            if (attempt == attempts || !RetryPolicy.isRetryable(lastError)) {
                break;
            }

            try {
                long delay = retryPolicy.delayMillis(attempt);
                log.info("Повтор подключения через {} мс", delay);
                retryPolicy.sleepBeforeRetry(attempt);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                break;
            }
        }

        throw new RetryExhaustedException(
                "Не удалось подключиться к БД за " + attempts + " попыт(ки). "
                        + "Проверьте, что контейнер PostgreSQL запущен и параметры db.* корректны.",
                attempts,
                lastError);
    }

    private boolean isValid(Connection connection) {
        try {
            int timeoutSeconds = (int) Math.max(1, settings.validationTimeoutMs() / 1000);
            return connection.isValid(timeoutSeconds);
        } catch (SQLException e) {
            return false;
        }
    }

    /**
     * Выполняет операцию с автоматическим закрытием соединения.
     */
    public <T> T withConnection(String operation, SqlWork<T> work) {
        try (Connection connection = getConnection()) {
            return work.apply(connection);
        } catch (SQLException e) {
            throw SqlExceptionTranslator.translate(e, operation);
        }
    }

    /**
     * Выполняет операцию в транзакции. При исключении выполняется rollback.
     */
    public <T> T inTransaction(String operation, SqlWork<T> work) {
        Connection connection = getConnection();
        boolean previousAutoCommit;
        try {
            previousAutoCommit = connection.getAutoCommit();
        } catch (SQLException e) {
            closeQuietly(connection);
            throw SqlExceptionTranslator.translate(e, operation);
        }

        try {
            connection.setAutoCommit(false);
            T result = work.apply(connection);
            connection.commit();
            return result;
        } catch (SQLException e) {
            rollbackQuietly(connection, e);
            throw SqlExceptionTranslator.translate(e, operation);
        } catch (RuntimeException | Error e) {
            rollbackQuietly(connection, e);
            throw e;
        } finally {
            try {
                connection.setAutoCommit(previousAutoCommit);
            } catch (SQLException ignored) {
                // Соединение уже может быть невалидным - Hikari его утилизирует
            }
            closeQuietly(connection);
        }
    }

    /**
     * Проверка доступности БД (SELECT 1).
     */
    public String ping() {
        return withConnection("ping", connection -> {
            try (Statement statement = connection.createStatement();
                 ResultSet resultSet = statement.executeQuery("SELECT version()")) {
                resultSet.next();
                return resultSet.getString(1);
            }
        });
    }

    private void rollbackQuietly(Connection connection, Throwable cause) {
        try {
            connection.rollback();
        } catch (SQLException rollbackError) {
            cause.addSuppressed(rollbackError);
        }
    }

    private void closeQuietly(Connection connection) {
        try {
            connection.close();
        } catch (SQLException ignored) {
            // ничего не делаем: соединение возвращается пулу/утилизируется
        }
    }

    @Override
    public void close() {
        dataSource.close();
        log.info("Пул соединений закрыт");
    }

    /**
     * Функциональный интерфейс операции с БД, допускающей SQLException.
     */
    @FunctionalInterface
    public interface SqlWork<T> {
        T apply(Connection connection) throws SQLException;
    }
}
````

````java
package org.example.sto.db;

import java.net.ConnectException;
import java.net.SocketException;
import java.net.SocketTimeoutException;
import java.sql.SQLException;
import java.sql.SQLRecoverableException;
import java.sql.SQLTransientConnectionException;

/**
 * Политика повторных попыток с экспоненциальной задержкой.
 *
 * <p>Повторяются только временные сбои: потеря/недоступность соединения,
 * перезапуск сервера БД, сериализационные ошибки и deadlock. Ошибки
 * целостности, синтаксиса и аутентификации не повторяются.
 */
public final class RetryPolicy {

    @FunctionalInterface
    public interface Sleeper {
        void sleep(long millis) throws InterruptedException;
    }

    private final int maxAttempts;
    private final long initialDelayMs;
    private final double multiplier;
    private final long maxDelayMs;
    private final Sleeper sleeper;

    public RetryPolicy(int maxAttempts, long initialDelayMs, double multiplier, long maxDelayMs) {
        this(maxAttempts, initialDelayMs, multiplier, maxDelayMs, Thread::sleep);
    }

    public RetryPolicy(int maxAttempts, long initialDelayMs, double multiplier, long maxDelayMs, Sleeper sleeper) {
        if (maxAttempts < 1) {
            throw new IllegalArgumentException("maxAttempts должен быть >= 1");
        }
        if (initialDelayMs < 0 || maxDelayMs < 0) {
            throw new IllegalArgumentException("Задержки не могут быть отрицательными");
        }
        if (multiplier < 1.0) {
            throw new IllegalArgumentException("multiplier должен быть >= 1.0");
        }
        this.maxAttempts = maxAttempts;
        this.initialDelayMs = initialDelayMs;
        this.multiplier = multiplier;
        this.maxDelayMs = maxDelayMs;
        this.sleeper = sleeper;
    }

    public int maxAttempts() {
        return maxAttempts;
    }

    /**
     * Задержка перед повтором после указанной неудачной попытки (нумерация с 1).
     */
    public long delayMillis(int failedAttempt) {
        if (failedAttempt < 1) {
            return initialDelayMs;
        }
        double delay = initialDelayMs * Math.pow(multiplier, failedAttempt - 1L);
        return Math.min(maxDelayMs, Math.max(0L, (long) delay));
    }

    public void sleepBeforeRetry(int failedAttempt) throws InterruptedException {
        long delay = delayMillis(failedAttempt);
        if (delay > 0) {
            sleeper.sleep(delay);
        }
    }

    /**
     * Определяет, является ли сбой временным и безопасным для повтора.
     */
    public static boolean isRetryable(Throwable throwable) {
        Throwable current = throwable;
        int depth = 0;
        while (current != null && depth++ < 10) {
            if (current instanceof SQLTransientConnectionException
                    || current instanceof SQLRecoverableException) {
                return true;
            }
            if (current instanceof SQLException sqlException) {
                String state = sqlException.getSQLState();
                if (state != null && (state.startsWith("08")
                        || "57P03".equals(state)
                        || "40001".equals(state)
                        || "40P01".equals(state))) {
                    return true;
                }
            }
            if (current instanceof ConnectException
                    || current instanceof SocketException
                    || current instanceof SocketTimeoutException) {
                return true;
            }
            current = current.getCause();
````

## 13. `src/main/java/org/example/sto/db/SqlExceptionTranslator.java`

Строк: 106

````java
package org.example.sto.db;

import org.example.sto.exception.ConnectionUnavailableException;
import org.example.sto.exception.DataIntegrityException;
import org.example.sto.exception.DatabaseException;
import org.example.sto.exception.DuplicateKeyException;
import org.example.sto.exception.ForeignKeyViolationException;
import org.example.sto.exception.QueryException;
import org.example.sto.exception.TransientDatabaseException;

import java.sql.SQLException;

/**
 * Преобразует {@link SQLException} в предметные исключения приложения.
 *
 * <p>Классы SQLState:
 * <ul>
 *     <li>08xxx, 57P03 — потеря/недоступность соединения;</li>
 *     <li>23xxx — нарушение целостности (23505 — UNIQUE, 23503 — FK, 23502 — NOT NULL);</li>
 *     <li>22xxx — ошибки данных (22001 — превышение длины);</li>
 *     <li>28xxx — аутентификация;</li>
 *     <li>42xxx — синтаксис SQL и несуществующие объекты;</li>
 *     <li>40001, 40P01 — временные сбои транзакций.</li>
 * </ul>
 */
public final class SqlExceptionTranslator {

    private SqlExceptionTranslator() {
    }

    public static DatabaseException translate(SQLException exception, String operation) {
        SQLException source = effectiveException(exception);
        String state = source.getSQLState();
        String details = trim(source.getMessage());

        if (state != null) {
            if (state.startsWith("08") || "57P03".equals(state)) {
                return new ConnectionUnavailableException(
                        "База данных недоступна. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if ("40001".equals(state) || "40P01".equals(state)) {
                return new TransientDatabaseException(
                        "Транзакция прервана из-за конкуренции (" + state + "). Операция: " + operation,
                        operation, state, exception);
            }
            if ("23505".equals(state)) {
                return new DuplicateKeyException(
                        "Нарушено ограничение уникальности. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if ("23503".equals(state)) {
                return new ForeignKeyViolationException(
                        "Нарушено ограничение внешнего ключа. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if (state.startsWith("23")) {
                return new DataIntegrityException(
                        "Нарушено ограничение целостности данных. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if ("22001".equals(state)) {
                return new DataIntegrityException(
                        "Превышена допустимая длина строкового значения. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if (state.startsWith("28")) {
                return new ConnectionUnavailableException(
                        "Ошибка аутентификации при подключении к БД. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
            if (state.startsWith("42")) {
                return new QueryException(
                        "Ошибка SQL-запроса. Операция: " + operation + ". Детали: " + details,
                        operation, state, exception);
            }
        }

        return new DatabaseException(
                "Ошибка выполнения операции с БД: " + operation + ". Детали: " + details,
                operation, state, exception);
    }

    /**
     * PostgreSQL может складывать исходную ошибку в цепочку nextException.
     */
    private static SQLException effectiveException(SQLException exception) {
        SQLException current = exception;
        SQLException result = exception;
        while (current != null) {
            if (current.getSQLState() != null) {
                result = current;
            }
            current = current.getNextException();
        }
        return result;
    }

    private static String trim(String message) {
        if (message == null || message.isBlank()) {
            return "без описания";
        }
        String oneLine = message.replace('\n', ' ').replace('\r', ' ').trim();
        return oneLine.length() > 300 ? oneLine.substring(0, 300) + "..." : oneLine;
    }
}
````

````java
/**
 * Доступ к таблицам roles/users. Проверка пароля выполняется в PostgreSQL
 * функцией crypt() из расширения pgcrypto, поэтому хеш не загружается в Java.
 */
public final class UserRepository {

    private final ConnectionManager connectionManager;

    public UserRepository(ConnectionManager connectionManager) {
        this.connectionManager = connectionManager;
    }

    public Optional<User> authenticate(String username, char[] password) {
        String passwordText = new String(password);
        try {
            String finalPassword = passwordText;
            return connectionManager.withConnection("users.authenticate", connection -> {
                String sql = """
                        SELECT u.id, u.username, u.full_name, r.name AS role
                        FROM users u
                        JOIN roles r ON r.id = u.role_id
                        WHERE u.username = ?
                          AND u.password_hash = crypt(?, u.password_hash)
                        """;
                try (PreparedStatement statement = connection.prepareStatement(sql)) {
                    statement.setString(1, username);
                    statement.setString(2, finalPassword);
                    try (ResultSet resultSet = statement.executeQuery()) {
                        return resultSet.next() ? Optional.of(mapUser(resultSet)) : Optional.empty();
                    }
                }
            });
        } finally {
            java.util.Arrays.fill(password, '\0');
        }
    }

    public List<User> findAll() {
        return connectionManager.withConnection("users.findAll", connection -> {
            String sql = """
                    SELECT u.id, u.username, u.full_name, r.name AS role
                    FROM users u

                    ORDER BY u.id
                    """;
````

````java
 * заранее, чтобы выдавать понятные сообщения пользователю.

public final class OrderService {

    private final ClientRepository clientRepository;

        this.clientRepository = clientRepository;
        this.carRepository = carRepository;

    public int createOrder(int clientId, int carId, Integer masterId, String description) {
        if (clientRepository.findById(clientId).isEmpty()) {
            throw new EntityNotFoundException("Клиент с id=" + clientId + " не найден");
        }
        var car = carRepository.findById(carId)
                .orElseThrow(() -> new EntityNotFoundException("Автомобиль с id=" + carId + " не найден"));
        if (car.clientId() != clientId) {
            throw new ValidationException(
                    "Автомобиль " + car.plateNum() + " принадлежит другому клиенту (id=" + car.clientId() + ")");
        }
        if (description == null || description.isBlank()) {
            throw new ValidationException("Описание неисправности обязательно");
        }
        if (description.trim().length() > 2000) {
            throw new ValidationException("Описание не должно превышать 2000 символов");
        }
        if (masterId != null) {
            Master master = masterRepository.findById(masterId)
                    .orElseThrow(() -> new EntityNotFoundException("Мастер с id=" + masterId + " не найден"));
            if (!master.active()) {
                throw new ValidationException("Мастер " + master.fullName() + " неактивен");
            }

        if (CLOSED_STATUSES.contains(order.statusName())) {
````

````sql
-- 6.1. Создание заявки
CREATE OR REPLACE FUNCTION create_order(
    p_client_id   INTEGER,
    p_car_id      INTEGER,
    p_master_id   INTEGER,
    p_description TEXT
)
RETURNS INTEGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_order_id INTEGER;
BEGIN
    IF NOT EXISTS (SELECT 1 FROM clients WHERE id = p_client_id) THEN
        RAISE EXCEPTION 'Клиент с id=% не найден', p_client_id;
    END IF;

    IF NOT EXISTS (SELECT 1 FROM cars WHERE id = p_car_id) THEN
        RAISE EXCEPTION 'Автомобиль с id=% не найден', p_car_id;
    END IF;

    INSERT INTO orders (client_id, car_id, master_id, status_id, description)
    VALUES (p_client_id, p_car_id, p_master_id, 1, p_description)
    RETURNING id INTO v_order_id;

    RETURN v_order_id;
END;
$$;

-- 6.2. Добавление услуги в заявку
CREATE OR REPLACE FUNCTION add_service_to_order(
    p_order_id   INTEGER,
    p_service_id INTEGER,
    p_quantity   INTEGER DEFAULT 1
)
RETURNS VOID
LANGUAGE plpgsql
AS $$
DECLARE
    v_price NUMERIC(10,2);
BEGIN
    SELECT price INTO v_price FROM services WHERE id = p_service_id;
    IF v_price IS NULL THEN
        RAISE EXCEPTION 'Услуга с id=% не найдена', p_service_id;
    END IF;

    INSERT INTO order_services (order_id, service_id, quantity, price_at)
    VALUES (p_order_id, p_service_id, p_quantity, v_price);
END;
$$;

-- 6.3. Закрытие заявки
CREATE OR REPLACE FUNCTION close_order(p_order_id INTEGER)
RETURNS VOID
LANGUAGE plpgsql
AS $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM orders WHERE id = p_order_id) THEN
        RAISE EXCEPTION 'Заявка с id=% не найдена', p_order_id;
    END IF;

    UPDATE orders
    SET status_id = (SELECT id FROM order_statuses WHERE name = 'завершена')
    WHERE id = p_order_id;
END;
$$;

-- 6.4. Расчёт стоимости заявки
CREATE OR REPLACE FUNCTION calculate_order_total(p_order_id INTEGER)
RETURNS NUMERIC
LANGUAGE plpgsql
AS $$
DECLARE
    v_total NUMERIC(10,2);
BEGIN
    SELECT COALESCE(SUM(price_at * quantity), 0)
    INTO v_total
    FROM order_services
    WHERE order_id = p_order_id;

    RETURN v_total;
END;
$$;

-- 6.5. Получение заявок клиента
CREATE OR REPLACE FUNCTION get_client_orders(p_client_id INTEGER)
RETURNS TABLE (
    order_id   INTEGER,
    car_plate  VARCHAR,
    status     VARCHAR,
    total_cost NUMERIC,
    created_at TIMESTAMP
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT o.id,
           cr.plate_num,
           os.name::VARCHAR,
           COALESCE(SUM(osv.price_at * osv.quantity), 0),
           o.created_at
    FROM orders o
    JOIN cars cr           ON cr.id = o.car_id
    JOIN order_statuses os ON os.id = o.status_id
    LEFT JOIN order_services osv ON osv.order_id = o.id
    WHERE o.client_id = p_client_id
    GROUP BY o.id, cr.plate_num, os.name, o.created_at
    ORDER BY o.created_at DESC;
END;
$$;

CREATE OR REPLACE FUNCTION trg_set_closed_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    IF NEW.status_id = (SELECT id FROM order_statuses WHERE name = 'завершена')
       AND NEW.closed_at IS NULL THEN
        NEW.closed_at := CURRENT_TIMESTAMP;
    END IF;

    IF NEW.status_id <> (SELECT id FROM order_statuses WHERE name = 'завершена') THEN
        NEW.closed_at := NULL;
    END IF;

    RETURN NEW;
END;
$$;

BEFORE UPDATE OF status_id ON orders
FOR EACH ROW

$$;
````
